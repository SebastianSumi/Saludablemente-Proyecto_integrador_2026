/**
 * Cabecera de una actividad del programa de salud.
 * Hibernate la crea/actualiza en SLB_ACTIVIDADES con ddl-auto: update.
 */
@Entity
@Table(name = "ACTIVIDAD", schema = "SLB_ACTIVIDADES")
@Getter
@Setter
@NoArgsConstructor
public class Actividad {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Column(name = "ID_ACTIVIDAD")
    private Long idActividad;

    @Column(name = "NOMBRE", nullable = false, length = 100)
    private String nombre;

    @Column(name = "DESCRIPCION", length = 300)
    private String descripcion;

    @Column(name = "FECHA_ACTIVIDAD", nullable = false)
    private LocalDate fechaActividad;

    @Column(name = "HORA_INICIO", nullable = false)
    private LocalTime horaInicio;

    @Column(name = "HORA_FIN", nullable = false)
    private LocalTime horaFin;

    @Column(name = "LUGAR", nullable = false, length = 100)
    private String lugar;

    @Column(name = "ESTADO", nullable = false, length = 20)
    private String estado;   // PROGRAMADA / EN_CURSO / FINALIZADA / CANCELADA

    @Column(name = "ID_USUARIO_CREADOR", nullable = false)
    private Long idUsuarioCreador;

    @Column(name = "ID_USUARIO_COORDINADOR", nullable = false)
    private Long idUsuarioCoordinador;

    @Column(name = "PUNTAJE_SALUD")
    private Integer puntajeSalud;

    // ============================================================
    // Campos para Asistencias (FASE A)
    // ============================================================

    @Column(name = "LATITUD_OFICIAL", precision = 10, scale = 8)
    private BigDecimal latitudOficial;

    @Column(name = "LONGITUD_OFICIAL", precision = 11, scale = 8)
    private BigDecimal longitudOficial;

    @Column(name = "TOLERANCIA_GEOCERCA_METROS")
    private Integer toleranciaGeocercaMetros;

    @Column(name = "REQUIERE_CHECK_IN_OUT")
    private Boolean requiereCheckInOut;

    @Column(name = "DURACION_MINUTOS")
    private Integer duracionMinutos;

    @Column(name = "PERIODO", length = 10)
    private String periodo;

    @Column(name = "PERMITE_AUTOSERVICIO")
    private Boolean permiteAutoservicio;

    @Column(name = "PERMITE_ESCANEO_COORDINADOR")
    private Boolean permiteEscaneoCoordinador;

    @Column(name = "CIERRE_ASISTENCIAS_REALIZADO")
    private Boolean cierreAsistenciasRealizado;
}

    private Long idPersona;

    private LocalDateTime fechaInscripcion;

     * true si fue inscrito via "inscribir a todos".

    @Column(name = "nivel_severidad", nullable = false, length = 15)

    @Column(name = "mensaje_resumen", length = 300)
    private String mensajeResumen;

    @Column(name = "estado", nullable = false, length = 15)
    private EstadoAlerta estado;

    @Column(name = "fecha_generacion", nullable = false)
    private LocalDateTime fechaGeneracion;

    @Column(name = "fecha_atencion")
    private LocalDateTime fechaAtencion;

    @Column(name = "id_usuario_atiende")
    private Long usuarioAtiendeId;

    @Column(name = "observacion_atencion", length = 300)
    private String observacionAtencion;

            estado = EstadoAlerta.PENDIENTE;

package upeu.edu.pe.Saludabemente.alertaclinica.alerta.repository;

import org.springframework.data.domain.Sort;
import org.springframework.data.jpa.repository.EntityGraph;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;
import upeu.edu.pe.Saludabemente.alertaclinica.alerta.dto.AlertaAgregado;
import upeu.edu.pe.Saludabemente.alertaclinica.alerta.dto.AlertaClinicaResumen;
import upeu.edu.pe.Saludabemente.alertaclinica.alerta.entity.AlertaClinica;
import upeu.edu.pe.Saludabemente.alertaclinica.alerta.entity.EstadoAlerta;
import upeu.edu.pe.Saludabemente.alertaclinica.alerta.entity.SeveridadAlerta;
import upeu.edu.pe.Saludabemente.alertaclinica.alerta.entity.TipoEvaluacionOrigen;

import java.time.LocalDateTime;
import java.util.List;
import java.util.Optional;

public interface AlertaClinicaRepository extends JpaRepository<AlertaClinica, Long> {

    @Override
    @EntityGraph(attributePaths = "detalles")
    Optional<AlertaClinica> findById(Long id);

    /**
     * Idempotencia: buscar si ya existe alerta para la misma evaluacion de origen.
     *

    /**
     * Listado administrativo con filtros combinados.
     */
    @EntityGraph(attributePaths = "detalles")
    @Query("""
        SELECT a FROM AlertaClinica a
        WHERE (:estado IS NULL OR a.estado = :estado)
          AND (:severidad IS NULL OR a.nivelSeveridad = :severidad)
          AND (:personaId IS NULL OR a.personaId = :personaId)
          AND (:tipoOrigen IS NULL OR a.tipoEvaluacionOrigen = :tipoOrigen)
          AND (:desde IS NULL OR a.fechaGeneracion >= :desde)
          AND (:hasta IS NULL OR a.fechaGeneracion <= :hasta)
        """)
    List<AlertaClinica> buscar(
            @Param("estado") EstadoAlerta estado,
            @Param("severidad") SeveridadAlerta severidad,
            @Param("personaId") Long personaId,
            @Param("tipoOrigen") TipoEvaluacionOrigen tipoOrigen,
            @Param("desde") LocalDateTime desde,
            @Param("hasta") LocalDateTime hasta,
            Sort sort
    );

    /**
     * Proyeccion de resumen (mas liviana, sin cargar detalles).
     */
    @Query("""
        SELECT new upeu.edu.pe.Saludabemente.alertaclinica.alerta.dto.AlertaClinicaResumen(
            a.idAlerta, a.personaId, a.tipoEvaluacionOrigen, a.puntajeTotal,
            a.nivelSeveridad, a.estado, a.fechaGeneracion, SIZE(a.detalles))
        FROM AlertaClinica a
        WHERE (:estado IS NULL OR a.estado = :estado)
          AND (:desde IS NULL OR a.fechaGeneracion >= :desde)
          AND (:hasta IS NULL OR a.fechaGeneracion <= :hasta)
        """)
    List<AlertaClinicaResumen> buscarResumen(
            @Param("estado") EstadoAlerta estado,
            @Param("desde") LocalDateTime desde,
            @Param("hasta") LocalDateTime hasta,
            Sort sort
    );

    /**
     * Agregados para el reporte.
     * COALESCE evita null cuando no hay filas.
     */
    @Query("""
        SELECT new upeu.edu.pe.Saludabemente.alertaclinica.alerta.dto.AlertaAgregado(
            COUNT(a),
            COALESCE(SUM(CASE WHEN a.nivelSeveridad = 'CRITICA' THEN 1 ELSE 0 END), 0L),
            COALESCE(SUM(CASE WHEN a.estado = 'ATENDIDA' THEN 1 ELSE 0 END), 0L))
        FROM AlertaClinica a
        WHERE (:desde IS NULL OR a.fechaGeneracion >= :desde)
          AND (:hasta IS NULL OR a.fechaGeneracion <= :hasta)
        """)
    AlertaAgregado agregados(
            @Param("desde") LocalDateTime desde,
            @Param("hasta") LocalDateTime hasta
    );

    Optional<AlertaClinica> findByEvaluacionOrigenIdAndTipoEvaluacionOrigen(
            Long evaluacionOrigenId, TipoEvaluacionOrigen tipo);

    boolean existsByIdAlertaAndPersonaIdAndEstado(
            Long idAlerta, Long personaId, EstadoAlerta estado);
}

/**
 * Cabecera transaccional del mÃ³dulo Aptitud FÃ­sica.
 * Contiene el snapshot inmutable de la persona al momento de iniciar la prueba.
 * Los diagnÃ³sticos se calculan en {@code MetricasAptitud} y se persisten aquÃ­.
 */
@Entity
@Table(name = "EVALUACION", schema = "SLB_APTITUD_FISICA")
@Getter
@Setter
@NoArgsConstructor
@AllArgsConstructor
@Builder
public class EvaluacionAptitud extends AuditableEntity {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    @Column(name = "id_evaluacion")
    private Long id;

    // ---- Hard-IDs (sin FK fÃ­sica â€” patrÃ³n Modulith) ----
    @Column(name = "id_persona", nullable = false)
    private Long idPersona;

    @Column(name = "id_periodo", nullable = false)
    private Long idPeriodo;

    // ---- Snapshot inmutable (copiado de PersonaSnapshotClinicoDTO) ----
    @Column(name = "documento_identidad_al_momento", length = 15, nullable = false)
    private String documentoIdentidadAlMomento;

    @Column(name = "nombres_completos_al_momento", length = 150, nullable = false)
    private String nombresCompletosAlMomento;

    @Column(name = "edad_al_momento", nullable = false, columnDefinition = "NUMBER(3)")
    private Integer edadAlMomento;

    @Column(name = "sexo_al_momento", length = 1, nullable = false)
    private String sexoAlMomento;

    @Column(name = "area_trabajo_al_momento", length = 100)
    private String areaTrabajoAlMomento;

    // ---- Ciclo de vida ----
    @Column(name = "fecha_evaluacion", nullable = false)
    private LocalDate fechaEvaluacion;

    @Enumerated(EnumType.STRING)
    @Column(name = "estado_evaluacion", length = 30, nullable = false)
    @Builder.Default
    private EstadoEvaluacion estadoEvaluacion = EstadoEvaluacion.PENDIENTE;

    @Enumerated(EnumType.STRING)
    @Column(name = "dictamen_global", length = 50)
    private DictamenGlobal dictamenGlobal;

    @Column(name = "observaciones_restricciones", length = 500)
    private String observacionesRestricciones;

    @Column(name = "fitness_score", precision = 5, scale = 2)
    private BigDecimal fitnessScore;

    // ---- Relaciones satÃ©lite 1:1 (mismo mÃ³dulo) ----
    @OneToOne(mappedBy = "evaluacion", fetch = FetchType.LAZY,
            cascade = CascadeType.ALL, orphanRemoval = true)
    private DetalleFuerzaResistencia detalleFuerza;

    @OneToOne(mappedBy = "evaluacion", fetch = FetchType.LAZY,
            cascade = CascadeType.ALL, orphanRemoval = true)
    private DetalleCardiovascular detalleCardio;
}
package upeu.edu.pe.Saludabemente.aptitudfisica.evaluacion.repository;

import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.jpa.repository.EntityGraph;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;
import upeu.edu.pe.Saludabemente.aptitudfisica.evaluacion.dto.resumen.AptitudAgregadoDto;
import upeu.edu.pe.Saludabemente.aptitudfisica.evaluacion.entity.EvaluacionAptitud;
import upeu.edu.pe.Saludabemente.aptitudfisica.evaluacion.enums.EstadoEvaluacion;

import java.util.List;
import java.util.Optional;

public interface EvaluacionAptitudRepository extends JpaRepository<EvaluacionAptitud, Long> {

    /** Evita N+1: trae cabecera + fuerza + cardio en una sola query LEFT JOIN. */
    @EntityGraph(attributePaths = {"detalleFuerza", "detalleCardio"})
    @Query("SELECT e FROM EvaluacionAptitud e WHERE e.id = :id")
    Optional<EvaluacionAptitud> findByIdConDetalle(@Param("id") Long id);

    boolean existsByIdPersonaAndIdPeriodo(Long idPersona, Long idPeriodo);

    @EntityGraph(attributePaths = {"detalleFuerza", "detalleCardio"})
    List<EvaluacionAptitud> findByIdPersonaOrderByFechaEvaluacionDesc(Long idPersona);

    /** BÃºsqueda dinÃ¡mica: todos los filtros son opcionales. */
    @Query("""
            SELECT e FROM EvaluacionAptitud e
            WHERE (:idPeriodo IS NULL OR e.idPeriodo = :idPeriodo)
              AND (:estado    IS NULL OR e.estadoEvaluacion = :estado)
              AND (:area      IS NULL OR LOWER(e.areaTrabajoAlMomento) LIKE LOWER(CONCAT('%', :area, '%')))
              AND (:texto     IS NULL OR LOWER(e.nombresCompletosAlMomento) LIKE LOWER(CONCAT('%', :texto, '%'))
                                      OR LOWER(e.documentoIdentidadAlMomento) LIKE LOWER(CONCAT('%', :texto, '%')))
            """)
    Page<EvaluacionAptitud> buscar(
            @Param("idPeriodo") Long idPeriodo,
            @Param("estado") EstadoEvaluacion estado,
            @Param("area") String area,
            @Param("texto") String texto,
            Pageable pageable);

    /**
     * MÃ©tricas agregadas por perÃ­odo para el dashboard.
     * Una sola query â€” evita 6 COUNT separados.
     */
    @Query("""
            SELECT new upeu.edu.pe.Saludabemente.aptitudfisica.evaluacion.dto.resumen.AptitudAgregadoDto(
                COUNT(e),
                COUNT(CASE WHEN e.estadoEvaluacion = upeu.edu.pe.Saludabemente.aptitudfisica.evaluacion.enums.EstadoEvaluacion.PENDIENTE THEN 1 END),
                COUNT(CASE WHEN e.estadoEvaluacion = upeu.edu.pe.Saludabemente.aptitudfisica.evaluacion.enums.EstadoEvaluacion.COMPLETA  THEN 1 END),
                COUNT(CASE WHEN e.estadoEvaluacion = upeu.edu.pe.Saludabemente.aptitudfisica.evaluacion.enums.EstadoEvaluacion.ANULADA   THEN 1 END),
                COUNT(CASE WHEN e.dictamenGlobal = upeu.edu.pe.Saludabemente.aptitudfisica.evaluacion.enums.DictamenGlobal.APTO THEN 1 END),
                COUNT(CASE WHEN e.dictamenGlobal = upeu.edu.pe.Saludabemente.aptitudfisica.evaluacion.enums.DictamenGlobal.APTO_CON_RESTRICCIONES THEN 1 END),
                COUNT(CASE WHEN e.dictamenGlobal = upeu.edu.pe.Saludabemente.aptitudfisica.evaluacion.enums.DictamenGlobal.NO_APTO THEN 1 END)
            )
            FROM EvaluacionAptitud e
            WHERE e.idPeriodo = :idPeriodo
            """)
    AptitudAgregadoDto obtenerAgregados(@Param("idPeriodo") Long idPeriodo);
}

    @Enumerated(EnumType.STRING)
    @Column(name = "escaneado_por", length = 15)

    private BigDecimal latitud;

    @Column(name = "longitud", precision = 10, scale = 7)
    private BigDecimal longitud;

    @Column(name = "distancia_geocerca_m", precision = 8, scale = 2)
    private BigDecimal distanciaGeocercaM;

    @Column(name = "flag_sospechoso", nullable = false)
    private Boolean flagSospechoso;

    @Column(name = "dispositivo_id", length = 60)
    private String dispositivoId;

    @Enumerated(EnumType.STRING)
    @Column(name = "sync_status", nullable = false, length = 10)
    private SyncStatus syncStatus;

    @Column(name = "retry_count", nullable = false)
    private Integer retryCount;

    @Column(name = "last_sync_attempt")
    private LocalDateTime lastSyncAttempt;

    @Column(name = "sync_error", length = 255)
    private String syncError;

    @Column(name = "id_usuario_registro")
    private Long idUsuarioRegistro;

    @Column(name = "id_qr_actividad")
    private Long idQrActividad;

    @Column(name = "codigo_qr_usado", length = 80)
    private String codigoQrUsado;

    @Column(name = "created_at", nullable = false, updatable = false)
    private LocalDateTime createdAt;

    @Column(name = "updated_at", nullable = false)
    private LocalDateTime updatedAt;

    @PrePersist
    public void prePersist() {
        LocalDateTime now = LocalDateTime.now();
        if (idLocal == null) {
            idLocal = UUID.randomUUID().toString();
        }
        if (createdAt == null) createdAt = now;
        updatedAt = now;
        if (flagSospechoso == null) flagSospechoso = false;
        if (retryCount == null) retryCount = 0;
        if (syncStatus == null) syncStatus = SyncStatus.SYNCED;
    }

    @PreUpdate
    public void preUpdate() {
        updatedAt = LocalDateTime.now();
    }
}

    @Column(name = "ID_TEAM")

    @Setter
    @NoArgsConstructor

        private Long idNoticia;
        private Long idTeam;

        public boolean equals(Object o) {
            if (this == o) return true;
            if (!(o instanceof NoticiaTeamId that)) return false;
            return Objects.equals(idNoticia, that.idNoticia)
                    && Objects.equals(idTeam, that.idTeam);
        }

        @Override
        public int hashCode() {
            return Objects.hash(idNoticia, idTeam);
        }
    }
}

@Entity

    @Column(name = "id_periodo", nullable = false, length = 10)

    @Column(name = "documento_identidad", length = 8, nullable = false)

    private String sexoAlMomento;

    @Column(name = "area_trabajo_al_momento", length = 100)
    private String areaTrabajoAlMomento;

    // ---- Ciclo de vida ----
    @Column(name = "fecha_evaluacion", nullable = false)
    private LocalDate fechaEvaluacion;

    @Enumerated(EnumType.STRING)
    @Column(name = "estado_evaluacion", nullable = false, length = 20)
    private EstadoEvaluacion estadoEvaluacion;

    @Column(name = "observaciones", length = 500)
    private String observaciones;

    // ---- Detalles (FK vive en los detalles) ----
    @OneToOne(mappedBy = "evaluacion", cascade = CascadeType.ALL,
            orphanRemoval = true, fetch = FetchType.LAZY)
    private DetalleAntropometrico detalleAntropometrico;

    @OneToOne(mappedBy = "evaluacion", cascade = CascadeType.ALL,
            orphanRemoval = true, fetch = FetchType.LAZY)
    private DetalleBioquimico detalleBioquimico;

    // ---- Helpers de consistencia bidireccional ----
    public void asignarAntropometrico(DetalleAntropometrico detalle) {
        if (detalle != null) {
            detalle.setEvaluacion(this);
        }
        this.detalleAntropometrico = detalle;
    }

    public void asignarBioquimico(DetalleBioquimico detalle) {
        if (detalle != null) {
            detalle.setEvaluacion(this);
        }
        this.detalleBioquimico = detalle;
    }

    @PrePersist
    void inicializarDefaults() {
        if (fechaEvaluacion == null) {
            fechaEvaluacion = LocalDate.now();
        }
        if (estadoEvaluacion == null) {
            estadoEvaluacion = EstadoEvaluacion.PENDIENTE;
        }
    }
}
package upeu.edu.pe.Saludabemente.nutricional.evaluacion.repository;

import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.jpa.repository.EntityGraph;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Modifying;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;
import upeu.edu.pe.Saludabemente.nutricional.evaluacion.entity.EvaluacionNutricional;
import upeu.edu.pe.Saludabemente.nutricional.evaluacion.enums.EstadoEvaluacion;

import java.util.Collection;
import java.util.List;
import java.util.Optional;

public interface EvaluacionRepository extends JpaRepository<EvaluacionNutricional, Long> {

    // ===================================================================
    // Consultas puntuales
    // ===================================================================

    @Override
    @EntityGraph(attributePaths = {"detalleAntropometrico", "detalleBioquimico"})
    Optional<EvaluacionNutricional> findById(Long id);

    // ===================================================================
    // BÃºsqueda paginada con filtros tolerantes a nulos
    // ===================================================================

    @EntityGraph(attributePaths = {"detalleAntropometrico"})
    @Query("""
            SELECT e FROM EvaluacionNutricional e
            WHERE (:estado IS NULL OR e.estadoEvaluacion = :estado)
              AND (:periodo IS NULL OR e.idPeriodo = :periodo)
              AND (:idPersona IS NULL OR e.idPersona = :idPersona)
            """)
    Page<EvaluacionNutricional> buscar(
            @Param("estado") EstadoEvaluacion estado,
            @Param("periodo") String periodo,
            @Param("idPersona") Long idPersona,
            Pageable pageable);

    // ===================================================================
    // Historial por persona
    // ===================================================================

    @EntityGraph(attributePaths = {"detalleAntropometrico", "detalleBioquimico"})
    List<EvaluacionNutricional> findByIdPersonaOrderByFechaEvaluacionDesc(Long idPersona);

    // ===================================================================
    // Unicidad semestral
    // ===================================================================

    @Query("""
            SELECT COUNT(e) > 0 FROM EvaluacionNutricional e
            WHERE e.idPersona = :idPersona
              AND e.idPeriodo = :idPeriodo
              AND e.estadoEvaluacion IN :estados
            """)
    boolean existsActivaPorPersonaYPeriodo(
            @Param("idPersona") Long idPersona,
            @Param("idPeriodo") String idPeriodo,
            @Param("estados") Collection<EstadoEvaluacion> estados);

    // ===================================================================
    // Evaluaciones incompletas
    // ===================================================================

    @EntityGraph(attributePaths = {"detalleAntropometrico", "detalleBioquimico"})
    @Query("""
            SELECT e FROM EvaluacionNutricional e
            WHERE e.idPeriodo = :periodo
              AND e.estadoEvaluacion = :estado
            """)
    List<EvaluacionNutricional> findIncompletasByPeriodo(
            @Param("periodo") String periodo,
            @Param("estado") EstadoEvaluacion estado);

    // ===================================================================
    // Conteos y mutaciones
    // ===================================================================

    long countByEstadoEvaluacion(EstadoEvaluacion estado);

    @Modifying(clearAutomatically = true)
    @Query("""
            UPDATE EvaluacionNutricional e
               SET e.estadoEvaluacion = :estado
             WHERE e.idEvaluacion = :id
            """)
    int softDelete(@Param("id") Long id,
                   @Param("estado") EstadoEvaluacion estado);

    // ===================================================================
    // Reportes
    // ===================================================================

    @EntityGraph(attributePaths = {"detalleBioquimico"})
    @Query("""
            SELECT e FROM EvaluacionNutricional e
             JOIN e.detalleBioquimico b
            WHERE e.idPeriodo = :periodo
              AND b.dxBioquimico LIKE 'ALTERADO%'
            """)
    List<EvaluacionNutricional> findBioquimicosAlteradosByPeriodo(
