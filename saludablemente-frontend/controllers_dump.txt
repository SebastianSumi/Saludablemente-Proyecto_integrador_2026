package upeu.edu.pe.Saludabemente.actividades.inscripcion.controller;

import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;
import upeu.edu.pe.Saludabemente.actividades.inscripcion.dto.request.InscribirPersonasRequest;
import upeu.edu.pe.Saludabemente.actividades.inscripcion.service.InscripcionService;

import java.util.List;
import java.util.Map;

@Tag(name = "Actividades - Inscripciones")
@RestController
@RequestMapping("/api/v1/actividades/{idActividad}/inscripciones")
@RequiredArgsConstructor
public class InscripcionController {

    private final InscripcionService inscripcionService;

    @Operation(summary = "Inscribir a TODAS las personas activas (reemplaza inscritos previos)")
    @PostMapping("/todos")
    public ResponseEntity<Map<String, Object>> inscribirATodos(@PathVariable Long idActividad) {
        int total = inscripcionService.inscribirATodos(idActividad);
        return ResponseEntity.ok(Map.of(
                "idActividad", idActividad,
                "inscritos", total,
                "modo", "TODOS"
        ));
    }

    @Operation(summary = "Inscribir personas especificas (agrega a los existentes)")
    @PostMapping
    public ResponseEntity<Map<String, Object>> inscribirPersonas(
            @PathVariable Long idActividad,
            @Valid @RequestBody InscribirPersonasRequest req) {
        int agregados = inscripcionService.inscribirPersonas(idActividad, req.idsPersonas());
        return ResponseEntity.ok(Map.of(
                "idActividad", idActividad,
                "agregados", agregados,
                "modo", "ESPECIFICO"
        ));
    }

    @Operation(summary = "Quitar un inscrito")
    @DeleteMapping("/{idPersona}")
    public ResponseEntity<Void> quitar(
            @PathVariable Long idActividad,
            @PathVariable Long idPersona) {

        boolean eliminado = inscripcionService.quitarInscrito(idActividad, idPersona);

        if (!eliminado) {
            // No existia el inscrito: 404
            return ResponseEntity.notFound().build();
        }

        // Se elimino correctamente: 204
        return ResponseEntity.noContent().build();
    }

    @Operation(summary = "Listar IDs de inscritos")
    @GetMapping
    public ResponseEntity<List<Long>> listar(@PathVariable Long idActividad) {
        return ResponseEntity.ok(inscripcionService.listarInscritos(idActividad));
    }

    @Operation(summary = "Contar inscritos")
    @GetMapping("/count")
    public ResponseEntity<Map<String, Long>> contar(@PathVariable Long idActividad) {
        return ResponseEntity.ok(Map.of("total", inscripcionService.contarInscritos(idActividad)));
    }
}

import upeu.edu.pe.Saludabemente.exception.BusinessRuleException;
import upeu.edu.pe.Saludabemente.metas.meta.dto.CheckInRequestDTO;
import upeu.edu.pe.Saludabemente.metas.meta.dto.ProgresoMetaResponseDTO;
import upeu.edu.pe.Saludabemente.metas.meta.service.AuditoriaMetasService;
import upeu.edu.pe.Saludabemente.metas.meta.service.CheckInService;
import upeu.edu.pe.Saludabemente.metas.meta.service.NotificacionMetasService;

import upeu.edu.pe.Saludabemente.seguridad.api.metas.MetasUsuarioApi;
import upeu.edu.pe.Saludabemente.seguridad.api.metas.UsuarioActualDTO;

import java.util.List;

@Slf4j
@Tag(name = "Check-ins")
@RestController
@RequestMapping("/api/v1/metas/checkins")
@RequiredArgsConstructor
public class CheckInController {

    private final CheckInService checkInService;
    private final NotificacionMetasService notificacionMetas;
    private final AuditoriaMetasService auditoriaMetas;
    private final MetasUsuarioApi seguridadApi;

    @Operation(summary = "Registra un avance sobre una meta")
    @PostMapping
    public ResponseEntity<ProgresoMetaResponseDTO> registrar(
            @Valid @RequestBody CheckInRequestDTO dto) {

        if (!seguridadApi.tieneAlgunRol("Colaborador", "Administrador")) {
            throw new BusinessRuleException("No autorizado para registrar check-ins");
        }
        UsuarioActualDTO usuario = seguridadApi.obtenerUsuarioActual();

        ProgresoMetaResponseDTO resultado = checkInService.registrar(dto, usuario.email());

        try {
            auditoriaMetas.registrarCheckIn(
                    usuario.idUsuario(),
                    resultado.idMeta(),
                    null,
                    resultado.modalidad().name(),
                    resultado.valorActual() == null ? null : resultado.valorActual().toString());
        } catch (RuntimeException ex) {
            log.warn("Fallo al auditar check-in de meta {}: {}",
                    resultado.idMeta(), ex.getMessage());
        }

        if (resultado.estado() == EstadoMeta.COMPLETADA) {
            try {
                notificacionMetas.notificarMetaCompletada(
                        resultado.idPersona(),
                        resultado.idMeta(),
                        resultado.descripcion());
            } catch (RuntimeException ex) {
                log.warn("Fallo al notificar meta completada {}: {}",
                        resultado.idMeta(), ex.getMessage());
            }
        }

        return ResponseEntity.ok(resultado);
    }

    @Operation(summary = "Historial de check-ins de una meta")
    @GetMapping("/meta/{idMeta}")
    public ResponseEntity<List<ProgresoMetaResponseDTO>> historial(@PathVariable Long idMeta) {

        if (!seguridadApi.tieneAlgunRol("Colaborador", "Administrador", "Evaluador")) {
            throw new BusinessRuleException("No autorizado para consultar historial");
        }
        return ResponseEntity.ok(checkInService.historial(idMeta));
    }
}

@RestController
@RequestMapping("/api/v1/evaluaciones-nutricionales")
@RequiredArgsConstructor
public class EvaluacionController {

    private final EvaluacionService evaluacionService;

    // ===================================================================
    // GET /api/v1/evaluaciones-nutricionales
    // ===================================================================

    private static final Set<String> CAMPOS_ORDENABLES = Set.of(
            "idEvaluacion",
            "fechaEvaluacion",
            "idPeriodo",
            "estadoEvaluacion",
            "idPersona"
    );

    @Operation(summary = "Lista evaluaciones con filtros opcionales y paginaciÃ³n")
    @ApiResponse(responseCode = "200", description = "Listado paginado de evaluaciones")
    @GetMapping
    public ResponseEntity<Page<EvaluacionResumen>> buscar(
            @Parameter(description = "Estado de la evaluaciÃ³n")
            @RequestParam(required = false) EstadoEvaluacion estado,

            @Parameter(description = "Periodo semestral (ej. 2026-1)")
            @RequestParam(required = false) String periodo,

            @Parameter(description = "ID de la persona")
            @RequestParam(required = false) Long idPersona,

            @PageableDefault(size = 20)
            Pageable pageable) {

        Pageable pageableSeguro = sanearPageable(pageable);
        return ResponseEntity.ok(
                evaluacionService.buscar(estado, periodo, idPersona, pageableSeguro));
    }

    /**
     * Ignora propiedades de ordenamiento que no sean campos reales de la entidad.
     * Swagger UI envÃ­a {@code sort=string} por defecto, y eso rompe Spring Data.
     * Si ningÃºn campo es vÃ¡lido, aplica el orden por defecto: fechaEvaluacion DESC.
     */
    private Pageable sanearPageable(Pageable pageable) {
        Sort ordenSeguro = Sort.by(Sort.Direction.DESC, "fechaEvaluacion");

        if (pageable.getSort().isSorted()) {
            Sort filtrado = Sort.unsorted();
            for (Sort.Order order : pageable.getSort()) {
                if (CAMPOS_ORDENABLES.contains(order.getProperty())) {
                    filtrado = filtrado.and(
                            Sort.by(order.getDirection(), order.getProperty()));
                }
            }
            if (filtrado.isSorted()) {
                ordenSeguro = filtrado;
            }
        }

        return PageRequest.of(pageable.getPageNumber(), pageable.getPageSize(), ordenSeguro);
    }

    // ===================================================================
    // GET /api/v1/evaluaciones-nutricionales/agregados
    // ===================================================================

    @Operation(summary = "MÃ©tricas agregadas para el dashboard")
    @ApiResponse(responseCode = "200", description = "Totales y promedios")

    public ResponseEntity<EvaluacionAgregado> agregados(

    }

    // ===================================================================

    // ===================================================================

    @Operation(summary = "Obtiene una evaluaciÃ³n completa por ID")
    @ApiResponses({

    // ===================================================================

    // ===================================================================

    @Operation(summary = "Historial cronolÃ³gico de evaluaciones de una persona")

    // ===================================================================

        return ResponseEntity.ok(evaluacionService.incompletasPorPeriodo(periodo));
    }

    // ===================================================================
    // POST /api/v1/evaluaciones-nutricionales
    // ===================================================================

    @Operation(summary = "Inicia una evaluaciÃ³n capturando el snapshot clÃ­nico")
    @ApiResponses({
            @ApiResponse(responseCode = "201", description = "EvaluaciÃ³n creada en PENDIENTE"),
            @ApiResponse(responseCode = "400", description = "Datos invÃ¡lidos"),
            @ApiResponse(responseCode = "404", description = "Persona no encontrada"),
            @ApiResponse(responseCode = "409", description = "Ya existe evaluaciÃ³n activa en el periodo")
    })
    @PostMapping
    public ResponseEntity<EvaluacionResponse> iniciar(
            @Valid @RequestBody EvaluacionInitRequest request) {

        EvaluacionResponse creada = evaluacionService.iniciar(request);
        return ResponseEntity.status(HttpStatus.CREATED).body(creada);
    }

    // PATCH /api/v1/evaluaciones-nutricionales/{id}/observaciones

    @ApiResponses({

@Tag(name = "Personal")

            "celular", "fechaNacimiento", "activo"

    private final PersonaService personaService;

        return ResponseEntity.ok(

    // -------------------------------------------------------------------

    @Operation(summary = "Inactiva una persona (baja logica)")
    @DeleteMapping("/{id}")
    @ResponseStatus(HttpStatus.NO_CONTENT)

        personaService.inactivar(id);

    // -------------------------------------------------------------------

    @Operation(summary = "Obtiene las preferencias de comunicacion de una persona")

    // -------------------------------------------------------------------
    // Helpers
    // -------------------------------------------------------------------

    private Sort construirSort(String ordenarPor, String direccion) {
        if (!CAMPOS_ORDENABLES.contains(ordenarPor)) {
            throw new IllegalArgumentException(
                    "Campo de ordenamiento no permitido: " + ordenarPor
                            + ". Permitidos: " + CAMPOS_ORDENABLES);
        }
        Sort.Direction dir = "DESC".equalsIgnoreCase(direccion)
                ? Sort.Direction.DESC
                : Sort.Direction.ASC;
        return Sort.by(dir, ordenarPor);
    }

    public ResponseEntity<Void> actualizarPassword(
            @PathVariable Long id,
            @Valid @RequestBody PasswordRequest request) {
        usuarioService.actualizarPassword(id, request);
        return ResponseEntity.noContent().build();
    }

    // -------------------------------------------------------------------
    // Inactivar / reactivar
    // -------------------------------------------------------------------

    @Operation(summary = "Inactivar una cuenta (baja lÃ³gica)")
    @ApiResponses({
            @ApiResponse(responseCode = "204", description = "Cuenta inactivada"),
            @ApiResponse(responseCode = "404", description = "Usuario no encontrado"),
            @ApiResponse(responseCode = "409", description = "La cuenta ya estÃ¡ inactiva")
    })
    @DeleteMapping("/{id}")
    public ResponseEntity<Void> inactivar(@PathVariable Long id) {
        usuarioService.inactivar(id);
        return ResponseEntity.noContent().build();
    }

    @Operation(summary = "Reactivar una cuenta inactivada")
    @ApiResponses({
            @ApiResponse(responseCode = "204", description = "Cuenta reactivada"),
            @ApiResponse(responseCode = "404", description = "Usuario no encontrado"),
            @ApiResponse(responseCode = "409", description = "La cuenta ya estÃ¡ activa")
    })
    @PostMapping("/{id}/reactivar")
    public ResponseEntity<Void> reactivar(@PathVariable Long id) {
        usuarioService.reactivar(id);
        return ResponseEntity.noContent().build();
    }
}
package upeu.edu.pe.Saludabemente.teams.controller;

import io.swagger.v3.oas.annotations.Operation;
import io.swagger.v3.oas.annotations.tags.Tag;
import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import org.springframework.data.domain.Sort;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.*;
import upeu.edu.pe.Saludabemente.teams.dto.request.TeamRequest;
import upeu.edu.pe.Saludabemente.teams.dto.response.TeamResponse;
import upeu.edu.pe.Saludabemente.teams.service.TeamService;

import java.util.List;

@Tag(name = "Teams")
@RestController
@RequestMapping("/api/v1/equipos")
@RequiredArgsConstructor
public class TeamController {

    private final TeamService teamService;

    @Operation(summary = "Lista equipos, opcionalmente solo los activos")
    @GetMapping
    public ResponseEntity<List<TeamResponse>> listar(
            @RequestParam(required = false) Boolean activo) {
        Sort sort = Sort.by(Sort.Direction.ASC, "nombre");
        return ResponseEntity.ok(teamService.listar(activo, sort));
    }

    @Operation(summary = "Consulta un equipo por id")
    @GetMapping("/{id}")
    public ResponseEntity<TeamResponse> obtener(@PathVariable Long id) {
        return ResponseEntity.ok(teamService.obtener(id));
    }

    @Operation(summary = "Registra un equipo nuevo")
    @PostMapping
    @ResponseStatus(HttpStatus.CREATED)
    public ResponseEntity<TeamResponse> crear(@Valid @RequestBody TeamRequest request) {
        return ResponseEntity.status(HttpStatus.CREATED).body(teamService.crear(request));
    }

    @Operation(summary = "Actualiza un equipo existente")
    @PutMapping("/{id}")
    public ResponseEntity<TeamResponse> actualizar(
            @PathVariable Long id,
            @Valid @RequestBody TeamRequest request) {
        return ResponseEntity.ok(teamService.actualizar(id, request));
    }

    @Operation(summary = "Inactiva un equipo (baja lÃ³gica)")
    @DeleteMapping("/{id}")
    @ResponseStatus(HttpStatus.NO_CONTENT)
    public void inactivar(@PathVariable Long id) {
        teamService.inactivar(id);
    }
}
